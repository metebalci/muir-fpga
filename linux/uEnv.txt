# The card is written once; everything after it arrives over Ethernet.
#
# U-Boot's preboot loads this from the card's FAT partition and imports it,
# then `uenvboot` imports it a second time and runs `uenvcmd`.  Two imports,
# and the order matters more than once below.
#
# WHY A FAILED FETCH IS THE CONTROL, AND NOT A FAILURE
#
# `default_bootcmd` is `run uenvboot; run cp_kernel2ram && bootm ${netstart}`
# --- a semicolon, not an `&&`.  So when the TFTP server is not answering, the
# `&&` chain in uenvcmd stops at the first fetch, uenvboot returns, and the
# card boots the stock `image.ub` with the BSP's own device tree.  Server off
# is the 512 MB control boot; server on is the 384 MB reserved boot; one card,
# nothing touched between them, and the two differ in exactly one 26 KB file on
# muirhost.  That is a better control than renaming a file on the card, which
# is what this used to require.  With `netretry` unset the failure is one
# attempt and not a loop (net/net.c, 669-685).
#
# WHY THERE IS NO ipaddr HERE
#
# preboot ends in an unconditional `dhcp`, on every boot, whatever this file
# says --- it is the last command of a string that was parsed before we were
# imported.  There is a DHCP server on this LAN, so it answers and sets ipaddr
# from yiaddr.  On a network without one, add ipaddr, netmask, and
# `bootpretryperiod=1000`: that last one does take effect on the same boot,
# because this import happens before the `dhcp` in the same command string
# (net/bootp.c line 716).
#
# WHY serverip SURVIVES THAT dhcp
#
# `CONFIG_BOOTP_SERVERIP` is defined (platform-auto.h line 107), and in
# net/bootp.c the whole serverip-and-bootfile block of `store_net_params()` is
# inside `#if !defined(CONFIG_BOOTP_SERVERIP)`.  A DHCP reply therefore sets
# ipaddr and leaves serverip alone.  The second import re-applies it after the
# dhcp in any case.  The compiled-in default is 172.16.119.130, which is not
# this network.
serverip=192.168.80.54

# WHY loadbootenv_addr MOVES, AND WHY ONLY THE SECOND IMPORT OBEYS IT
#
# preboot sets it to 0x1EE00000, which is 115 MiB inside the 128 MB the CADR
# owns --- in the spare region past the display.  Because this file is imported
# before `uenvboot` runs its own `load`, the value below is what the second
# load uses.  The first load already happened at 0x1EE00000 and nothing on the
# card can change that.  0x0f800000 is clear of the reservation, of the tree at
# 0x0f000000, and of the kernel at 0x10000000.
loadbootenv_addr=0x0f800000

# 1468 rather than the 512-byte default (RFC 2348).  This build has
# tftpblocksize and tftptimeout; it has no tftpwindowsize, which postdates it.
tftpblocksize=1468

# fdt_high=~0 leaves the tree where it was loaded instead of relocating it, and
#   initrd_high=~0 does the same for a ramdisk.  Tidiness, not safety: the
#   relocation ceiling is CONFIG_SYS_BOOTMAPSZ + bootm_low = 0x08000000 + 0,
#   256 MB below the reservation, so neither could have landed in it anyway.
#
# THEY ARE SET LATE, INSIDE uenvcmd, AND THAT IS DELIBERATE.  Set as plain
# variables here they would also apply to the fallback `bootm image.ub`, where
# fdt_high=~0 makes boot_relocate_fdt call fdt_set_totalsize(fdt, size +
# CONFIG_SYS_FDT_PAD) on the tree *inside the FIT*.  Measured on this image.ub:
# the fdt's data runs to file offset 3,778,390 and the ramdisk's data begins at
# 3,778,568, so a blob declared 26,262 + 12,288 bytes long overlaps the ramdisk
# by 12,110 and the memory and bootargs fixups are written into it.  Putting
# the two setenvs after the fetches means a card whose server is down boots the
# stock path with them unset, which is what makes the control a real control.
# The one window left is both fetches succeeding and bootz then failing, which
# means a corrupt zImage on the server and a console session either way.
uenvcmd=tftpboot 0x0f000000 system.dtb && tftpboot 0x10000000 zImage && setenv fdt_high 0xffffffff && setenv initrd_high 0xffffffff && bootz 0x10000000 - 0x0f000000
