# The mutation list for cadr-chaosnet: bugs `chaos_test` has to catch.
#
# `mutate.py` applies each record to a copy of this package's sources, builds
# the check against it, and runs it.  A mutation the check FAILS on is a check
# doing its job; one the check passes is reported as SURVIVED and the run is
# red.  A record whose `@old` no longer matches its file exactly once, or that
# does not compile, is BROKEN and also fails the run --- CLAUDE.md records two
# fabric mutations reported as surviving that had never been built.
#
# THE FORMAT is `mutations/list.txt`'s, so that anybody who has read that one
# can read this one:
#
#   @mutation  a kebab-case name; how a survivor is reported
#   @file      which of this package's core sources to mutate
#   @why       what the check is supposed to notice, printed when it does not
#   @note      prose, repeatable; printed with @why on a survivor
#   @old       the source to replace, exactly as it stands
#   @new       what to put there; empty deletes
#   @end
#
# WHERE THE LIST CAME FROM.  One record for every decision this program makes
# that a reader would otherwise have to take on trust, grouped by the layer it
# is in: the packet's own arithmetic, the register face's handshake with the
# fabric, and CHUDP's frame.  Those three layers are the whole program.
#
# **FORTY-NINE RECORDS WENT WHEN THE SERVICES DID.**  This list held four
# records for the connection protocol, three for the TIME, UPTIME and STATUS
# answers and forty-two for the FILE service.  A CADR has no file or time
# server in it, muir removed its own at `79c7590`, and so this program has
# none: the host is on the network.  A record for code that should not exist
# is not a record, so they went with the code rather than being kept as a
# list of bugs in a program nobody runs.
#
# **THE CHECK WORD HAS A REAL ANCHOR AND THAT IS WHY ITS RECORDS ARE WORTH
# HAVING.**  muir's `tests/chaos.rs` pins `135771` as the word the NETLIST
# BOARD itself produced for a particular twelve words, and says that of the
# 9401's polynomials, its two bit orders and its two seeds, only one
# arrangement gives it.  `chaos_test_packet` holds this C to that number.  So
# a mutation of the polynomial is not caught by a round trip agreeing with
# itself --- which it would, both ends being wrong together --- but by
# disagreeing with a measurement off real hardware.

@mutation the-check-word-polynomial-loses-a-tap
@file chaos_packet.c
@why the check word against 135771, which is what the netlist board produced for muir's twelve words
@note The 9401's CRC-16 is x^16 + x^15 + x^2 + 1.  Dropping the x^2 term
@note leaves a polynomial that is still a CRC and still round-trips against
@note itself, so only the board's own figure catches it.
@old
				next ^= 1u | (1u << 2) | (1u << 15);
@new
				next ^= 1u | (1u << 15);
@end

@mutation the-check-word-shifts-least-significant-bit-first
@file chaos_packet.c
@why the check word against 135771: the two 74165s at LMTBUF B12 and B13 shift a word out most-significant bit first, and the other order gives a different word
@note muir's packet.rs says this bit order is not read off a document --- it
@note is the one arrangement of the orders, seeds and polynomials the 9401
@note offers that reproduces the board's own output.
@old
		for (int k = 15; k >= 0; --k) {
@new
		for (int k = 0; k <= 15; ++k) {
@end

@mutation the-data-bytes-go-high-half-first
@file chaos_packet.c
@why the packet's words against muir's own literals: AIM-628 section 3.6 puts the first byte of a pair in the arithmetically LEAST significant position
@note "TIME" is 044524 044515 in muir's test and would be 052104 015515
@note swapped.  A round trip alone would not see it, because the reader here
@note would unswap what the writer swapped.
@old
			(uint16_t)(p->data[lo] |
				   ((hi < p->len ? (unsigned)p->data[hi] : 0u) << 8));
@new
			(uint16_t)(((unsigned)p->data[lo] << 8) |
				   (hi < p->len ? (unsigned)p->data[hi] : 0u));
@end

@mutation an-odd-byte-count-loses-its-last-byte
@file chaos_packet.c
@why a packet of five bytes: the data is a whole number of WORDS on the cable, so an odd count rounds UP and the spare half is padding
@note AIM-628 section 7: an odd last byte "sits in the low half with a
@note garbage padding byte in its high half".  Rounding down silently drops
@note the last byte of every odd-length packet, which is most text.
@old
	const unsigned data_words = ((unsigned)p->len + 1u) / 2u;
@new
	const unsigned data_words = (unsigned)p->len / 2u;
@end

@mutation the-check-word-is-computed-over-itself
@file chaos_packet.c
@why the frame's check word against 135771: the word covers the buffer and the source the hardware adds, and stops there
@old
	out[n + 2] = chaos_check_word(out, n + 2);
@new
	out[n + 2] = chaos_check_word(out, n + 1);
@end

@mutation a-frame-with-a-wrong-check-word-is-called-good
@file chaos_packet.c
@why a frame whose check word does not match must come back with check_ok clear, because that is the only thing between a corrupted packet and the transport acting on it
@note The interface has a CRC Error bit for exactly this and AIM-628 section
@note 5.1 meters it.  A parser that always says the frame is good turns every
@note line error into a packet the NCP believes.
@old
	out->check_ok = chaos_check_word(words, body + 2) == out->check;
@new
	out->check_ok = 1;
@end

@mutation the-data-count-is-not-held-against-the-frames-length
@file chaos_packet.c
@why a frame whose 12-bit count disagrees with the number of words it actually carries must be refused, not read past its end
@note AIM-628 section 5.1 meters incoming packets "rejected for a length that
@note is not a multiple of 16 bits" and its neighbors.  Without this the
@note parser reads the trailer as data, or data as the trailer.
@old
	if (body != CHAOS_PKT_HEADER_WORDS + (count + 1u) / 2u) {
@new
	if (body < CHAOS_PKT_HEADER_WORDS) {
@end

@mutation the-forwarding-count-eats-the-byte-count
@file chaos_packet.c
@why the count word is the forwarding count in bits 15:12 and the byte count in 11:0, and a mask of 017777 lets a forwarding count of 1 read as 4096 bytes
@old
	const unsigned count = words[1] & 07777u;
@new
	const unsigned count = words[1] & 017777u;
@end

# ---- the register face ----

@mutation the-frame-is-taken-without-telling-the-interface
@file chaos_face.c
@why CTL_TX_TAKE is what lets the machine's Transmit Done come up and its next packet go; without it the interface holds one frame for ever and the machine's Chaosnet stops after a single packet
@old
	f->write(f, CHAOS_CTL, CHAOS_CTL_TX_TAKE);
	return (int)len;
@new
	return (int)len;
@end

@mutation a-frame-the-interface-refused-is-reported-as-stored
@file chaos_face.c
@why a commit the fabric refused --- the machine had not emptied its incoming buffer --- must come back as 0 so the sender retransmits, and LOST moving is what says it was refused
@note STAT cannot tell the two apart: RX_BUSY is up either way, because a
@note stored frame leaves it up too.  So the test is the counter, which is
@note the interface's own four-bit Lost Count from AIM-628 section 7.
@old
	if (lost_after != lost_before)
@new
	if (lost_after == lost_before)
@end

# ---- Chaosnet over UDP ----

@mutation the-chudp-trailer-is-read-in-the-packets-own-order
@file chaos_udp.c
@why CHUDP is a MIXED frame: the packet's own words are little-endian and the hardware trailer's three are network order, and reading the trailer the other way puts the destination and source addresses byte-swapped
@note muir marks both orders unverified and says what would settle them, and
@note says the wrong order fails loudly rather than quietly.  This record is
@note what makes that true here: the byte-for-byte datagram in the check is
@note the one thing that would have to change with them.
@old
			get_word(CHUDP_TRAILER_ORDER, datagram + len - CHUDP_TRAILER + 2u * k);
@new
			get_word(CHUDP_PACKET_ORDER, datagram + len - CHUDP_TRAILER + 2u * k);
@end

@mutation an-unknown-chudp-version-is-read-as-this-one
@file chaos_udp.c
@why a datagram carrying a version this does not speak is refused WITH ITS NUMBER, not parsed as version 1
@note The protocol's author has said a version 2 may differ from version 1 in
@note nothing but byte order, so a version 2 peer read as a version 1 one
@note would exchange nonsense rather than fail --- which is the one failure
@note mode a wire protocol must not have.
@old
	if (datagram[0] != CHUDP_VERSION) {
@new
	if (datagram[0] == 0xFFu) {
@end

# ---- the route of last resort, and the table nobody writes ----
#
# muir removed `--chaos-udp-dynamic` and added `--chaos-udp-default-peer` at
# its own `d6eac6d`, and these records hold this program to the same six
# decisions.  Each is one the reader would otherwise have to take on trust:
# where a frame goes that no peer entry names, where a broadcast does not go,
# that the default peer does not stand in front of a named one, that a frame
# for this cable's own address goes nowhere, that a datagram is judged by what
# is in it and not by the socket it came off, and that nothing on this cable
# can be impersonated by a datagram claiming its address.

@mutation the-default-peer-is-never-reached
@file chaos_udp.c
@why a frame whose destination no peer entry names goes to the default peer, which is the route of last resort and the only routing this program has
@note A peer entry says that ONE address lives at ONE endpoint, so naming a
@note bridge as a peer does not let this board talk through it: before the
@note default peer, a frame for any other address was dropped.  Delete the
@note fallback and the board is back to talking only to hosts it was told
@note about by name, which is the fault the flag exists to fix.
@old
	if (sent == 0 && cable_dest != 0 && u->have_default) {
@new
	if (0 && cable_dest != 0 && u->have_default) {
@end

@mutation the-default-peer-is-handed-a-broadcast
@file chaos_udp.c
@why a broadcast goes to the named peers alone: they are stations on this machine's own cable, and the default peer is the way out to a wider network
@note Handing a bridge a broadcast would put this cable's broadcast on a
@note network it was never meant to reach.  muir decided this deliberately and
@note says so at the function; a record is what stops somebody deciding the
@note other way by accident, since the peers still get it either way and
@note nothing else would show.
@old
	if (sent == 0 && cable_dest != 0 && u->have_default) {
@new
	if (u->have_default && (cable_dest == 0 || sent == 0)) {
@end

@mutation the-default-peer-stands-in-front-of-a-named-one
@file chaos_udp.c
@why the default peer is where what is NOT named goes, not a route in front of the names: an endpoint a flag gave is where that address is reached, bridge or no bridge
@note The frame still reaches the peer, so the peer's end of the check sees
@note nothing wrong; what this catches is the copy the bridge should never
@note have been given.  A board that sent every frame to the bridge as well
@note would work and would leak every packet onto the wider network.
@old
	if (sent == 0 && cable_dest != 0 && u->have_default) {
@new
	if (cable_dest != 0 && u->have_default) {
@end

@mutation a-frame-for-this-cable-goes-out-over-udp
@file chaos_udp.c
@why a destination this cable already carries is reached on the cable and not over UDP, and it is not the bridge's business either
@note Without the default peer this was harmless: no peer entry claims the
@note machine's own address, so such a frame was dropped for want of anywhere
@note to go.  With a default peer there is always somewhere to go, so the
@note guard has to be real --- the bridge would otherwise be told about every
@note packet the machine addressed to itself.
@old
	if (cable_dest != 0 && cable_dest == u->local)
		return 0;
@new
	if (cable_dest != 0 && cable_dest == 0xFFFFu)
		return 0;
@end

@mutation a-datagram-from-an-endpoint-no-flag-named-is-turned-away
@file chaos_udp.c
@why a datagram is judged by what is IN the frame and never by the socket it came off, so a host no flag named is heard exactly as a named peer is
@note This is the door this program used to have and muir never did.  It was
@note defensible only while an endpoint could be learned, because then a host
@note that got in could also be answered; with nothing learned it refuses
@note traffic the machine's own transport would have to answer through a
@note bridge, whose datagrams come from the bridge and not from the host.
@old
		// The cable destination and the cable source the far side's
@new
		int named = 0;
		for (unsigned k = 0; k < u->npeers; ++k) {
			if (u->peers[k].where.sin_addr.s_addr == from.sin_addr.s_addr &&
			    u->peers[k].where.sin_port == from.sin_port)
				named = 1;
		}
		if (!named)
			continue;
		// The cable destination and the cable source the far side's
@end

@mutation a-datagram-may-claim-this-cables-own-address
@file chaos_udp.c
@why a datagram whose cable source is an address this process already carries is a forgery: the interface would take the frame for its own, Transmit Done and all
@note muir refuses it in the same place and for the same words.  This guard
@note did not have to bite while a stranger was turned away at the door,
@note because no datagram from an unnamed endpoint got this far; it is what
@note has to hold now that one does.
@old
		if (cable_source == 0 || cable_source == u->local) {
@new
		if (cable_source == 0) {
@end

@mutation a-bare-port-for-the-default-peer-is-not-the-loopback
@file chaos_udp.c
@why a bare port names the loopback at that port, which is what every endpoint flag here means by one
@note muir's own endpoint parser has the same rule and the same reason: a
@note bare port is a port on this host.  Binding the wildcard address as a
@note DESTINATION is not a destination at all, and a sendto against it goes
@note somewhere the person who typed the port did not name.
@old
		lands.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
@new
		lands.sin_addr.s_addr = htonl(INADDR_ANY);
@end

@mutation a-second-default-peer-is-taken
@file chaos_udp.c
@why there is one route of last resort, and a second endpoint for it is a typed statement contradicting a typed statement
@note The same rule as a second endpoint for one peer address, and the same
@note reason: guessing which was meant is worse than refusing.  Taking the
@note second silently would make the order of two lines in a file decide
@note where every unnamed frame goes.
@old
	if (u->have_default) {
		say("udp: a default peer twice; there is one route of last resort");
		return -1;
	}
@new
	if (0) {
		say("udp: a default peer twice; there is one route of last resort");
		return -1;
	}
@end

@mutation a-peer-flag-brings-its-own-cable
@file chaos_udp.c
@why the switches and the cable are two flags: a peer says who is ON the cable and is refused when there is no cable, rather than bringing one of its own
@note This is exactly what this program did before muir split the two at
@note `0851fa7`, and the fault it makes is quiet rather than loud: a run that
@note said who its file and time host was found itself on a network it had
@note not asked for, bound to a port nobody had named.  The default peer's
@note leg is left alone, so what this catches is the peer's alone.
@old
	if (have_cable)
		return NULL;
@new
	if (have_cable || npeers > 0)
		return NULL;
@end
