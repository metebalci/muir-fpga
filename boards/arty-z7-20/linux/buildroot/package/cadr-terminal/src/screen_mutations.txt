# The mutation list for cadr-terminal: bugs `screen_test.c` has to catch.
#
# `mutate.py` applies each record to a copy of this package's sources, builds
# the check against it, and runs it.  A mutation the check FAILS on is a check
# doing its job; one the check passes is reported as SURVIVED and the run is
# red.  A record whose `@old` no longer matches its file exactly once, or that
# does not compile, is BROKEN and also fails the run --- CLAUDE.md records two
# fabric mutations reported as surviving that had never been built.
#
# THE FORMAT is `mutations/list.txt`'s, so that anybody who has read that one
# can read this one:
#
#   @mutation  a kebab-case name; how a survivor is reported
#   @file      which of this package's sources to mutate
#   @why       what the check is supposed to notice, printed when it does not
#   @note      prose, repeatable; printed with @why on a survivor
#   @old       the source to replace, exactly as it stands
#   @new       what to put there; empty deletes
#   @end
#
# WHERE THE LIST CAME FROM.  The five the slice was asked for --- the bit
# order reversed within a word, the line stride off by one word, black and
# white swapped, an update rectangle off by one, and a frame served from a
# stale copy --- and then one for each decision this program makes that a
# reader would otherwise have to take on trust: the two encodings and which is
# chosen, the byte order a viewer asked for, the rows a diff calls changed,
# the whole-screen interval, and the two ways a viewer is refused.
#
# **THE BIT ORDER AND THE STRIDE ARE WRITTEN TWICE AND BOTH ARE MUTATED.**
# `screen_geom.h`'s `screen_lit` is the rule, and `screen_server.c`'s
# `row_byte` is the same rule again as a byte at a time, which is what a
# whole-width Raw rectangle actually goes through --- muir has the same pair
# (`Frame::shows_white` beside `Pixels::put_row`) and holds them to each other
# in `tests/terminal.rs`.  Mutating one and not the other is what says the
# check reaches both: the Raw pass of the anchors catches `row_byte` and the
# RRE pass catches `screen_lit`, because RRE asks the rule per pixel and Raw
# asks the table per byte.  A check that ran only one encoding would let half
# of this list through.

@mutation bit-order-reversed-within-a-word
@file screen_geom.h
@why the anchors: word 0 bit 0 is the top-left pixel, and reversed it is pixel 31
@note This is the classic failure of a program like this one --- a picture
@note served mirrored --- and it is invisible to any check that builds its
@note screen with the same rule it reads it with.  The anchors are literal
@note coordinates worked out by hand from muir src/simpletv.rs:255.
@old
	return (int)((words[bit / 32u] >> (bit % 32u)) & 1u);
@new
	return (int)((words[bit / 32u] >> (31u - bit % 32u)) & 1u);
@end

@mutation line-stride-a-word-short-in-the-pixel-rule
@file screen_geom.h
@why the anchors under RRE: word 24 bit 0 is the first pixel of line 1, and at 23 words a line it is not
@note RRE asks the rule for every pixel, so this is the pass that reaches it;
@note a whole-width Raw rectangle goes through `row_byte` instead.
@old
	const unsigned bit = y * SCREEN_WORDS_PER_LINE * 32u + x;
@new
	const unsigned bit = y * (SCREEN_WORDS_PER_LINE - 1u) * 32u + x;
@end

@mutation line-stride-a-word-short-in-the-encoder
@file screen_server.c
@why the anchors under Raw: the byte table is what a whole-width rectangle goes through, and it walks the wrong row
@old
	uint32_t word = f->words[y * f->words_per_line + k / 4];
@new
	uint32_t word = f->words[y * (f->words_per_line - 1) + k / 4];
@end

@mutation black-and-white-swapped
@file screen_geom.h
@why the anchors: with MODE BOW clear a lit bit shows WHITE, and swapped the whole screen comes back inverted
@note muir src/simpletv.rs:268.  The anchors run in both directions of BOW,
@note so a program that had simply lost the BOW term would also be caught.
@old

@new

@end

# ---------------------------------------------------------------------------
# THE KEYBOARD AND THE MOUSE.  `input_keys.c` is `muir::terminal::keyboard`'s
# state machine in C and `input_face.c` is the register face under it, and
# what these break is the handful of decisions that are not data: the frame
# bits, which half of a word says up, the shift worked around a character,
# a modifier being a key of its own, the prefix, the dropped release, the
# pointer's first event, and the room asked for before a word is written.
#
# **THE TABLE ITSELF IS NOT MUTATED AND SHOULD NOT BE.**  `input_keymap.h` is
# generated from muir's `keyboard.rs` and `default.keys`, so a mutation of it
# would be a mutation of a copy: what holds it is that the generator reads
# muir, and what holds THAT is the eighteen named keys and fourteen modifiers
# the check pins by position.

@mutation keys-the-frame-bits-are-left-out
@file input_keys.h
@why the word's top eight bits are what say which keyboard sent it
@note `keyboard::FRAME` is `0o37 << 19 | SOURCE << 16`: bits 23-19 "Reserved,
@note must be 1's" and 18-16 the source ID of the NEW keyboard, so every
@note up-down word has `word >> 16` equal to 0o371.  Without them the card's
@note high half reads as zero and the machine is being typed at by a
@note keyboard that does not exist --- and the high half is exactly where
@note microcode 323's Unibus channel looks first.  Caught at "the frame bits
@note of a key word".
@old
#define KEY_FRAME ((037u << 19) | (1u << 16))
@new
#define KEY_FRAME 0u
@end

@mutation keys-the-up-bit-says-down
@file input_keys.h
@why bit 8 is "1=key up, 0=key down", and a machine that decodes from the stream believes it
@note There are no modifier bits in a word on this keyboard: the machine
@note follows the stream of positions and works shift out for itself.  So a
@note key whose release looks like a second press is a key the machine never
@note sees come up --- and on a modifier that is a Control held for the rest
@note of the run.  Caught at "'a' down and up", where the two words come out
@note the same.
@old
	return KEY_FRAME | (up ? KEY_UP : 0u) | (position & 0177u);
@new
	return KEY_FRAME | (up ? 0u : KEY_UP) | (position & 0177u);
@end

@mutation keys-the-shift-is-not-worked-around-a-character
@file input_keys.c
@why RFB sends the shifted keysym and the machine wants the position with Shift down
@note `Keyboard::tap`'s whole reason: a viewer pressing shift and 1 sends
@note `Shift_L` and then `!`, and the Lisp Machine wants position 0o121 with
@note the Shift key down.  Sending the position alone types `1` where the
@note viewer typed `!`, which is the most visible thing this program can get
@note wrong and the thing no round trip would catch.  Caught at "'!' with no
@note shift held: the shift worked around it".
@old
	if (wants_shift && !held) {
		push(k, key_up_down(shift, 0));
		push(k, key_up_down(p, 0));
		push(k, key_up_down(p, 1));
		push(k, key_up_down(shift, 1));
	} else if (!wants_shift && held) {
@new
	if (wants_shift && !held) {
		push(k, key_up_down(p, 0));
		push(k, key_up_down(p, 1));
	} else if (!wants_shift && held) {
@end

@mutation keys-a-shift-the-viewer-holds-is-not-lifted
@file input_keys.c
@why the other half of the same decision, and the half a shifted screen never reaches
@note A viewer holding Shift and pressing a key whose UNSHIFTED plane is the
@note one wanted --- `(` with shift down, which is at 0o132 unshifted --- has
@note the shift lifted around the key and put back.  Leaving it down types
@note the shifted plane instead.  This is the branch the check reaches by
@note holding `Shift_L` and then sending a plain letter's keysym, which is
@note what a viewer with caps lock on actually sends.  Caught at "Shift_L and
@note 'A': the plane the viewer holds" --- the words are right there, so this
@note is caught by the count in "a character whose plane the viewer is not
@note holding" instead, where the held shift has to come up.
@old
		for (unsigned i = 0; i < m; ++i)
			push(k, key_up_down(up[i], 1));
		push(k, key_up_down(p, 0));
		push(k, key_up_down(p, 1));
		for (unsigned i = 0; i < m; ++i)
			push(k, key_up_down(up[i], 0));
@new
		push(k, key_up_down(p, 0));
		push(k, key_up_down(p, 1));
@end

@mutation keys-a-modifier-is-tapped-like-a-character
@file input_keys.c
@why a shifting key is HELD, and holding it is the only thing it does
@note `Keyboard::resolve` presses or releases a modifier at its own position
@note "and nothing more".  Tapping it instead sends the key down and up at
@note once, so a viewer holding Control and typing a letter sends Control
@note down, Control up, and then the letter --- and the machine, which
@note decodes from the stream, gets a plain letter every time.  Caught at
@note "Shift_L", where the two words come out as four.
@old

		if (down)
			press(k, (uint8_t)mp);
		else
			release(k, (uint8_t)mp);
		return;
	}
@new

			tap(k, (uint8_t)mp, 0);

@end

@mutation keys-every-release-is-dropped-and-not-only-a-tapped-ones
@file input_keys.c
@why a key the terminal did NOT send whole still owes the machine its release
@note `resolve`'s first branch drops the release of a key that has already
@note gone as a tap, and ONLY of such a key.  Dropping every release instead
@note leaves every key down for ever, which on a modifier is a Control the
@note machine goes on seeing for the rest of its run.  Caught at "Shift_L",
@note where the two words come out as one.
@note
@note **AND THE OBVIOUS MUTATION HERE IS EQUIVALENT, WHICH IS WORTH
@note RECORDING SO THAT NOBODY FILES IT AS A HOLE.**  Removing the early
@note RETURN and keeping the bookkeeping --- `if (!down) (void)take_tapped(k,
@note keysym);` --- survives, and not because the check is blind.  A tapped
@note key's position is not in `down`, and every path the release then falls
@note through to ends in `release()`, which does nothing for a position that
@note is not down.  So the `tapped` set's whole job is to skip work that
@note would be a no-op anyway, and the only live form is this one: the
@note branch taken for keys it was never meant for.
@old
	if (!down && take_tapped(k, keysym))
		return;
@new
	if (!down)
		return;
@end
